    <script>
        // Better seeded random number generator
        class SeededRandom {
            constructor(seed) {
                this.seed = seed;
            }
            
            next() {
                this.seed = (this.seed * 9301 + 49297) % 233280;
                return this.seed / 233280;
            }
        }
        
        function shuffleWithSeed(array, seed) {
            const rng = new SeededRandom(seed);
            const shuffled = [...array];
            for (let i = shuffled.length - 1; i > 0; i--) {
                const j = Math.floor(rng.next() * (i + 1));
                [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
            }
            return shuffled;
        }
        
        function generateSeed() {
            const randomSeed = Math.floor(Math.random() * 1000000);
            document.getElementById('seed').value = randomSeed;
        }
        
        function distribute() {
            // Get input values
            const valuesText = document.getElementById('values').value;
            const recipientsText = document.getElementById('recipients').value;
            const distributionType = document.getElementById('distribution-type').value;
            const seedInput = document.getElementById('seed').value;
            
            // Parse inputs
            const values = valuesText.split('\n').filter(v => v.trim() !== '');
            const recipients = recipientsText.split('\n').filter(r => r.trim() !== '');
            
            if (values.length === 0 || recipients.length === 0) {
                alert('Please enter both values and recipients!');
                return;
            }
            
            // Generate or use provided seed
            let seed;
            if (seedInput.trim() === '') {
                seed = Math.floor(Math.random() * 1000000);
                document.getElementById('seed').value = seed;
            } else {
                // Convert string to numeric seed
                let hash = 0;
                for (let i = 0; i < seedInput.length; i++) {
                    const char = seedInput.charCodeAt(i);
                    hash = ((hash << 5) - hash) + char;
                    hash = hash & hash; // Convert to 32-bit integer
                }
                seed = Math.abs(hash);
            }
            
            // Distribute based on selected method
            let distribution;
            switch(distributionType) {
                case 'equal':
                    distribution = distributeEqually(values, recipients, seed);
                    break;
                case 'random':
                    distribution = distributeRandomly(values, recipients, seed);
                    break;
                case 'roundrobin':
                    distribution = distributeRoundRobin(values, recipients);
                    break;
                default:
                    distribution = distributeRandomly(values, recipients, seed);
            }
            
            // Display results
            displayResults(distribution, seed);
        }
        
        function distributeEqually(values, recipients, seed) {
            const result = {};
            const valuesPerRecipient = Math.floor(values.length / recipients.length);
            const extraValues = values.length % recipients.length;
            
            // Shuffle recipients for fairness using seed
            const shuffledRecipients = shuffleWithSeed([...recipients], seed);
            
            // Shuffle values using different seed
            const shuffledValues = shuffleWithSeed([...values], seed + 1);
            
            let valueIndex = 0;
            for (let i = 0; i < shuffledRecipients.length; i++) {
                const recipient = shuffledRecipients[i];
                const count = valuesPerRecipient + (i < extraValues ? 1 : 0);
                
                result[recipient] = shuffledValues.slice(valueIndex, valueIndex + count);
                valueIndex += count;
            }
            
            return result;
        }
        
        function distributeRandomly(values, recipients, seed) {
            const result = {};
            recipients.forEach(r => result[r] = []);
            
            // Shuffle values using seed
            const shuffledValues = shuffleWithSeed([...values], seed);
            
            // Distribute randomly using the same RNG
            const rng = new SeededRandom(seed);
            shuffledValues.forEach((value) => {
                const recipientIndex = Math.floor(rng.next() * recipients.length);
                const recipient = recipients[recipientIndex];
                result[recipient].push(value);
            });
            
            return result;
        }
        
        function distributeRoundRobin(values, recipients) {
            const result = {};
            recipients.forEach(r => result[r] = []);
            
            let recipientIndex = 0;
            values.forEach(value => {
                const recipient = recipients[recipientIndex % recipients.length];
                result[recipient].push(value);
                recipientIndex++;
            });
            
            return result;
        }
        
        function displayResults(distribution, seed) {
            const resultDiv = document.getElementById('result');
            const shareBtn = document.getElementById('share-btn');
            resultDiv.style.display = 'block';
            shareBtn.style.display = 'inline-block';
            
            let html = '<h2>Distribution Results</h2>';
            html += `<p><strong>Shared Seed:</strong> <code>${document.getElementById('seed').value}</code> (Share this with others for consistent results)</p>`;
            
            let totalAssigned = 0;
            for (const [recipient, values] of Object.entries(distribution)) {
                html += `<p><strong>${recipient}:</strong> ${values.join(', ')} (${values.length} items)</p>`;
                totalAssigned += values.length;
            }
            
            html += `<p><strong>Total values distributed:</strong> ${totalAssigned} out of ${document.getElementById('values').value.split('\n').filter(v => v.trim() !== '').length}</p>`;
            
            resultDiv.innerHTML = html;
        }
        
        function shareResults() {
            const values = document.getElementById('values').value;
            const recipients = document.getElementById('recipients').value;
            const distributionType = document.getElementById('distribution-type').value;
            const seed = document.getElementById('seed').value;
            
            if (!values || !recipients) {
                alert('Please enter values and recipients first!');
                return;
            }
            
            // Create a simple text representation that can be shared
            const shareData = {
                values: values,
                recipients: recipients,
                type: distributionType,
                seed: seed
            };
            
            // For GitHub Pages, we can't modify URL directly, so we'll create a shareable instruction
            const instruction = `To get the same distribution:\n\n1. Go to: ${window.location.href.split('?')[0]}\n2. Enter these values:\n\nVALUES:\n${values}\n\nRECIPIENTS:\n${recipients}\n\n3. Set distribution type to: ${distributionType}\n4. Set seed to: ${seed}\n5. Click "Distribute Values"`;
            
            // Try to copy to clipboard
            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(instruction).then(() => {
                    showCopyNotification();
                }).catch(() => {
                    fallbackCopy(instruction);
                });
            } else {
                fallbackCopy(instruction);
            }
        }
        
        function showCopyNotification() {
            const notification = document.getElementById('copy-notification');
            notification.style.display = 'block';
            setTimeout(() => {
                notification.style.display = 'none';
            }, 3000);
        }
        
        function fallbackCopy(text) {
            const textArea = document.createElement('textarea');
            textArea.value = text;
            document.body.appendChild(textArea);
            textArea.select();
            try {
                document.execCommand('copy');
                showCopyNotification();
            } catch (err) {
                prompt('Copy the following text to share with your friends:', text);
            }
            document.body.removeChild(textArea);
        }
        
        // Load any preset data from URL parameters
        window.addEventListener('DOMContentLoaded', () => {
            const urlParams = new URLSearchParams(window.location.search);
            if (urlParams.has('seed')) {
                document.getElementById('seed').value = urlParams.get('seed');
            }
        });
