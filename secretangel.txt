    <script>
        // Seeded random number generator
        function seededRandom(seed) {
            let x = Math.sin(seed) * 10000;
            return x - Math.floor(x);
        }
        
        function shuffleWithSeed(array, seed) {
            const shuffled = [...array];
            for (let i = shuffled.length - 1; i > 0; i--) {
                const j = Math.floor(seededRandom(seed + i) * (i + 1));
                [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
            }
            return shuffled;
        }
        
        function generateSeed() {
            document.getElementById('seed').value = Math.floor(Math.random() * 1000000).toString();
        }
        
        function distribute() {
            // Get input values
            const valuesText = document.getElementById('values').value;
            const recipientsText = document.getElementById('recipients').value;
            const distributionType = document.getElementById('distribution-type').value;
            const seedInput = document.getElementById('seed').value;
            
            // Parse inputs
            const values = valuesText.split('\n').filter(v => v.trim() !== '');
            const recipients = recipientsText.split('\n').filter(r => r.trim() !== '');
            
            if (values.length === 0 || recipients.length === 0) {
                alert('Please enter both values and recipients!');
                return;
            }
            
            // Generate or use provided seed
            const seed = seedInput ? parseInt(seedInput) : Math.floor(Math.random() * 1000000);
            if (seedInput === '') {
                document.getElementById('seed').value = seed;
            }
            
            // Distribute based on selected method
            let distribution;
            switch(distributionType) {
                case 'equal':
                    distribution = distributeEqually(values, recipients, seed);
                    break;
                case 'random':
                    distribution = distributeRandomly(values, recipients, seed);
                    break;
                case 'roundrobin':
                    distribution = distributeRoundRobin(values, recipients);
                    break;
                default:
                    distribution = distributeRandomly(values, recipients, seed);
            }
            
            // Display results
            displayResults(distribution, seed);
        }
        
        function distributeEqually(values, recipients, seed) {
            const result = {};
            const valuesPerRecipient = Math.floor(values.length / recipients.length);
            const extraValues = values.length % recipients.length;
            
            // Shuffle recipients for fairness using seed
            const shuffledRecipients = shuffleWithSeed([...recipients], seed);
            
            // Shuffle values using seed
            const shuffledValues = shuffleWithSeed([...values], seed + 1); // Different seed for values
            
            let valueIndex = 0;
            for (let i = 0; i < shuffledRecipients.length; i++) {
                const recipient = shuffledRecipients[i];
                const count = valuesPerRecipient + (i < extraValues ? 1 : 0);
                
                result[recipient] = shuffledValues.slice(valueIndex, valueIndex + count);
                valueIndex += count;
            }
            
            return result;
        }
        
        function distributeRandomly(values, recipients, seed) {
            const result = {};
            recipients.forEach(r => result[r] = []);
            
            // Shuffle values using seed
            const shuffledValues = shuffleWithSeed([...values], seed);
            
            // Distribute randomly
            shuffledValues.forEach((value, index) => {
                const recipientIndex = Math.floor(seededRandom(seed + index) * recipients.length);
                const recipient = recipients[recipientIndex];
                result[recipient].push(value);
            });
            
            return result;
        }
        
        function distributeRoundRobin(values, recipients) {
            const result = {};
            recipients.forEach(r => result[r] = []);
            
            let recipientIndex = 0;
            values.forEach(value => {
                const recipient = recipients[recipientIndex % recipients.length];
                result[recipient].push(value);
                recipientIndex++;
            });
            
            return result;
        }
        
        function displayResults(distribution, seed) {
            const resultDiv = document.getElementById('result');
            const shareBtn = document.getElementById('share-btn');
            resultDiv.style.display = 'block';
            shareBtn.style.display = 'inline-block';
            
            let html = '<h2>Distribution Results</h2>';
            html += `<p><strong>Shared Seed:</strong> ${seed} (Share this with others for consistent results)</p>`;
            
            for (const [recipient, values] of Object.entries(distribution)) {
                html += `<p><strong>${recipient}:</strong> ${values.join(', ')}</p>`;
            }
            
            resultDiv.innerHTML = html;
            
            // Store the current state for sharing
            sessionStorage.setItem('distributionData', JSON.stringify({
                values: document.getElementById('values').value,
                recipients: document.getElementById('recipients').value,
                distributionType: document.getElementById('distribution-type').value,
                seed: seed,
                distribution: distribution
            }));
        }
        
        function shareResults() {
            const data = sessionStorage.getItem('distributionData');
            if (!data) {
                alert('No distribution data to share!');
                return;
            }
            
            // Create a shareable URL with parameters
            const { values, recipients, distributionType, seed } = JSON.parse(data);
            const params = new URLSearchParams({
                values: btoa(encodeURIComponent(values)),
                recipients: btoa(encodeURIComponent(recipients)),
                type: distributionType,
                seed: seed
            });
            
            const shareUrl = `${window.location.origin}${window.location.pathname}?${params.toString()}`;
            
            // Copy to clipboard
            navigator.clipboard.writeText(shareUrl).then(() => {
                alert('Shareable URL copied to clipboard! Send this to your friends.');
            }).catch(() => {
                // Fallback if clipboard API not available
                prompt('Copy this URL to share with your friends:', shareUrl);
            });
        }
        
        // Load data from URL parameters if present
        window.addEventListener('DOMContentLoaded', () => {
            const urlParams = new URLSearchParams(window.location.search);
            if (urlParams.has('values') && urlParams.has('recipients')) {
                try {
                    document.getElementById('values').value = decodeURIComponent(atob(urlParams.get('values')));
                    document.getElementById('recipients').value = decodeURIComponent(atob(urlParams.get('recipients')));
                    document.getElementById('distribution-type').value = urlParams.get('type') || 'equal';
                    document.getElementById('seed').value = urlParams.get('seed') || '';
                    
                    // Auto-distribute if all parameters are present
                    if (urlParams.has('seed')) {
                        distribute();
                    }
                } catch (e) {
                    console.error('Error loading from URL parameters:', e);
                }
            }
        });
